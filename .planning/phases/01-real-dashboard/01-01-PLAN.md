const [profileResult, subscriptionsResult, tutorsResult] = await Promise.all([
  supabase
    .from("profiles")
    .select("streak_count")
    .eq("id", user.id)
    .maybeSingle(),
  supabase
    .from("subscriptions")
    .select(`
      id,
      subscribed_at,
      classrooms (
        id,
        name,
        bio,
        tutor_id
      )
    `)
    .eq("student_id", user.id)
    .eq("status", "active")
    .order("subscribed_at", { ascending: false }),
  supabase
    .from("tutors")
    .select(`
      id,
      created_at,
      profiles (
        full_name,
        email
      )
    `)

]);
