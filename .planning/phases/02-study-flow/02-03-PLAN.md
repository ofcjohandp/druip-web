    ```typescript
    const handleAnswer = (selectedIndex: number) => {
    ```

    ```typescript
    import { View, StyleSheet } from 'react-native';
    import { useLocalSearchParams } from 'expo-router';
    import { LessonEngine } from '@/features/study/LessonEngine';
    import { COLORS } from '@/features/ui/theme';
    import { useQuery } from '@tanstack/react-query';
    import { supabase } from '@/lib/supabase';

    export default function LessonScreen() {
      const { lessonId, topicId } = useLocalSearchParams<{ lessonId: string; topicId: string }>();

      // Fetch lesson record for xpReward
      const { data: lesson } = useQuery({
        queryKey: ['lesson', lessonId],
        queryFn: async () => {
          const { data } = await supabase
            .from('lessons')
            .select('xp_reward, topic_id')
            .eq('id', lessonId)
            .single()
            .throwOnError();
          return data;
        },
        enabled: !!lessonId,
      });

      const resolvedTopicId = topicId || lesson?.topic_id || '';

      if (!lesson) return null; // Loading — LessonEngine handles its own loading state for questions

      return (
        <View style={styles.container}>
          <LessonEngine
            lessonId={lessonId}
            topicId={resolvedTopicId}
            xpReward={lesson.xp_reward}
          />
        </View>
      );
    }

    const styles = StyleSheet.create({
      container: { flex: 1, backgroundColor: COLORS.background },
    });
    ```

<threat_model>
| Threat | Severity | Mitigation |
|--------|----------|------------|
| Client-side answer manipulation (changing correct_option_index) | MEDIUM | For Phase 2 MVP, score is computed client-side and written via RLS-protected mutation. Server-side validation (trigger checking answers against DB) is Phase 3+ scope. Risk is acceptable: students cheating on their own progress hurts only themselves. |
| Answer option index out of bounds | LOW | `.slice(0, 4)` caps options; `correct_option_index` is validated at content seeding time. Runtime guard: if `correctIndex >= options.length`, treat as last option. |
| Race condition: double-tap on answer | LOW | `lockAnswer` has idempotent guard (`if (get().isLocked) return`); `disabled={isLocked}` on TouchableOpacity prevents re-render tap. |
| Back navigation to completed quiz session | MEDIUM | `router.replace` (not push) for lesson-complete navigation; `clearSession()` in useEffect cleanup on LessonEngine unmount |
</threat_model>
