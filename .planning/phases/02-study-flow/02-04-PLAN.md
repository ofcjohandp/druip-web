    ```typescript
    import { useQuery } from '@tanstack/react-query';
    import { supabase } from '@/lib/supabase';

    export function useProfile(userId: string | undefined) {
      return useQuery({
        queryKey: ['profile', userId],
        queryFn: async () => {
          const { data } = await supabase
            .from('profiles')
            .select('streak_count, total_xp')
            .eq('id', userId!)
            .single()
            .throwOnError();
          return data;
        },
        enabled: !!userId,
      });
    }
    ```

| Score tampering via route params | MEDIUM | Score/total are passed as route params from client-side computation. Server-side answer validation is Phase 3+ scope. For MVP, RLS ensures only the user can write their own attempts. Self-cheating has no competitive impact (no leaderboard). |
| XP inflation by replaying lesson completion | LOW | Each completion inserts a new `lesson_attempts` row; `user_lesson_progress` is upserted (idempotent). Total XP increment is not implemented in Phase 2 — that's Phase 4 (GAME-06). |
| Streak count read from profile without verification | LOW | Streak is read-only display on lesson-complete. Streak increment logic is Phase 4 (GAME-01). Display only, no write. |
| Schema push exposing migration secrets | LOW | `SUPABASE_ACCESS_TOKEN` is a deploy token, not a user secret. Used only in CI/manual push context. |
