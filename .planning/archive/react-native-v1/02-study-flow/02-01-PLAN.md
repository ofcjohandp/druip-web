    **`src/features/study/useLessonQuestions.ts`** (per D-14):

    TanStack Query hook. Fetch questions through sections join:
    ```typescript
    import { useQuery } from '@tanstack/react-query';
    import { supabase } from '@/lib/supabase';

    export function useLessonQuestions(lessonId: string) {
      return useQuery({
        queryKey: ['lesson-questions', lessonId],
        queryFn: async () => {
          const { data } = await supabase
            .from('questions')
            .select('*, sections!inner(lesson_id)')
            .eq('sections.lesson_id', lessonId)
            .eq('is_published', true)
            .order('order', { ascending: true })
            .limit(20)
            .throwOnError();
          return data ?? [];
        },
        enabled: !!lessonId,
        staleTime: 5 * 60 * 1000,
      });
    }
    ```

    Note: `.limit(20)` is the safety cap per RESEARCH.md Open Question 1.

    **`src/features/study/useCompleteLesson.ts`** (per D-12):
    ```typescript
    import { useMutation, useQueryClient } from '@tanstack/react-query';
    import { supabase } from '@/lib/supabase';

    export function useCompleteLesson() {
      const queryClient = useQueryClient();
      return useMutation({
        mutationFn: async ({ lessonId, userId, score, total }: {
          lessonId: string; userId: string; score: number; total: number;
        }) => {
          await supabase.from('lesson_attempts').insert({
            user_id: userId,
            lesson_id: lessonId,
            score,
            total_questions: total,
            status: 'completed',
            started_at: new Date().toISOString(),
            completed_at: new Date().toISOString(),
          }).throwOnError();

          await supabase.from('user_lesson_progress').upsert({
            user_id: userId,
            lesson_id: lessonId,
            completed: true,
            score,
            completed_at: new Date().toISOString(),
          }, { onConflict: 'user_id,lesson_id' }).throwOnError();
        },
        onSuccess: () => {
          queryClient.invalidateQueries({ queryKey: ['user-lesson-progress'] });
          queryClient.invalidateQueries({ queryKey: ['lessons'] });
        },
      });
    }
    ```

    All hooks: import `supabase` from `@/lib/supabase`, use `.throwOnError()` on every Supabase call.
